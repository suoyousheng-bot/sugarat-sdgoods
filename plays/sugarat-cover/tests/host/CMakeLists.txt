cmake_minimum_required(VERSION 3.19)
project(cover_host C)
set(CMAKE_C_STANDARD 11)
get_filename_component(FIRMWARE "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(APP "${FIRMWARE}/main/cover")
set(LVGL "${FIRMWARE}/components/lvgl")
set(WATCH "${FIRMWARE}/components/sugarat_watch")
file(GLOB_RECURSE LVGL_SOURCES CONFIGURE_DEPENDS "${LVGL}/src/*.c")
file(GLOB_RECURSE WATCH_SOURCES CONFIGURE_DEPENDS "${WATCH}/src/*.c")
if(APPLE AND EXISTS "/opt/homebrew/opt/openssl@3")
    set(OPENSSL_ROOT_DIR "/opt/homebrew/opt/openssl@3")
endif()
find_package(OpenSSL 3 REQUIRED)
add_library(cover-runtime STATIC ${LVGL_SOURCES} ${WATCH_SOURCES}
    "${APP}/cv_crypto.c" "${APP}/cv_store.c" "${APP}/cv_jpeg.c" "${APP}/cv_link.c"
    "${APP}/cv_copy.c" "${APP}/cv_gesture.c" "${APP}/cv_view.c")
target_include_directories(cover-runtime SYSTEM PUBLIC "${LVGL}" "${WATCH}/include" "${WATCH}/src")
target_include_directories(cover-runtime PUBLIC . "${APP}")
target_include_directories(cover-runtime SYSTEM PUBLIC "${FIRMWARE}/components/sdgoods_board/include")
target_compile_definitions(cover-runtime PUBLIC LV_CONF_INCLUDE_SIMPLE=1 LV_LVGL_H_INCLUDE_SIMPLE=1 _DARWIN_C_SOURCE _POSIX_C_SOURCE=200809L)
target_compile_options(cover-runtime PRIVATE -O1 -g -Wall -Wextra -Wno-unused-parameter -Wno-misleading-indentation -fsanitize=address,undefined -fno-omit-frame-pointer)
target_link_libraries(cover-runtime PUBLIC OpenSSL::Crypto m)
add_executable(cover-core-tests core_test.c)
target_link_libraries(cover-core-tests PRIVATE cover-runtime)
target_compile_options(cover-core-tests PRIVATE -Wall -Wextra -Werror -Wno-misleading-indentation -fsanitize=address,undefined -UNDEBUG)
target_link_options(cover-core-tests PRIVATE -fsanitize=address,undefined)
add_executable(cover-ui-tests ui_test.c)
target_link_libraries(cover-ui-tests PRIVATE cover-runtime)
target_compile_options(cover-ui-tests PRIVATE -Wall -Wextra -Werror -Wno-misleading-indentation -fsanitize=address,undefined -UNDEBUG)
target_link_options(cover-ui-tests PRIVATE -fsanitize=address,undefined)
if(ESP_IDF_ROOT)
    set(JSON_DIR "${ESP_IDF_ROOT}/components/json/cJSON")
    if(NOT EXISTS "${JSON_DIR}/cJSON.c")
        message(FATAL_ERROR "ESP_IDF_ROOT must contain the matching cJSON dependency")
    endif()
    add_executable(cover-http-bridge http_bridge.c "${JSON_DIR}/cJSON.c")
    target_include_directories(cover-http-bridge PRIVATE . "${JSON_DIR}")
    target_compile_options(cover-http-bridge PRIVATE -Wall -Wextra -Wno-misleading-indentation -fsanitize=address,undefined -UNDEBUG)
    target_link_options(cover-http-bridge PRIVATE -fsanitize=address,undefined)
    target_link_libraries(cover-http-bridge PRIVATE cover-runtime)
endif()
