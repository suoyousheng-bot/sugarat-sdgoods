cmake_minimum_required(VERSION 3.16)
project(card_host C)
get_filename_component(FIRMWARE "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(CARD "${FIRMWARE}/main/card")
set(BSP "${FIRMWARE}/components/sdgoods_board")
set(LVGL "${FIRMWARE}/components/lvgl")
set(WATCH "${FIRMWARE}/components/sugarat_watch")
set(JSON "$ENV{IDF_PATH}/components/json/cJSON")
if(NOT EXISTS "${JSON}/cJSON.c")
    message(FATAL_ERROR "Set IDF_PATH to the installed ESP-IDF 5.5.4 checkout")
endif()
if(APPLE AND EXISTS "/opt/homebrew/opt/openssl@3")
    set(OPENSSL_ROOT_DIR "/opt/homebrew/opt/openssl@3")
endif()
find_package(OpenSSL 3 REQUIRED)
file(GLOB_RECURSE LVGL_SRC "${LVGL}/src/*.c")
file(GLOB_RECURSE WATCH_SRC "${WATCH}/src/*.c")
add_executable(card-core core_test.c "${CARD}/card_data.c" "${CARD}/card_store.c" "${CARD}/card_crypto.c" "${CARD}/card_exchange.c" "${JSON}/cJSON.c")
target_include_directories(card-core PRIVATE "${CARD}" "${JSON}")
target_link_libraries(card-core PRIVATE OpenSSL::Crypto m)
add_executable(card-profile profile_cli.c "${CARD}/card_data.c" "${CARD}/card_store.c" "${CARD}/card_crypto.c" "${JSON}/cJSON.c")
target_include_directories(card-profile PRIVATE "${CARD}" "${JSON}")
target_link_libraries(card-profile PRIVATE OpenSSL::Crypto m)
add_executable(card-ui ui_test.c ${LVGL_SRC} ${WATCH_SRC} "${CARD}/card_data.c" "${CARD}/card_view.c" "${JSON}/cJSON.c"
    "${BSP}/src/sdgoods_tap.c" "${BSP}/src/sdgoods_swipe_back.c")
target_include_directories(card-ui PRIVATE . "${CARD}" "${BSP}/include" "${LVGL}" "${WATCH}/include" "${JSON}")
target_compile_definitions(card-ui PRIVATE LV_CONF_INCLUDE_SIMPLE)
target_link_libraries(card-ui PRIVATE m)
add_executable(card-http http_test.c "${CARD}/card_http.c" "${CARD}/card_data.c" "${CARD}/card_crypto.c" "${JSON}/cJSON.c")
target_include_directories(card-http PRIVATE . "${CARD}" "${JSON}")
target_compile_definitions(card-http PRIVATE CARD_HOST_HTTP)
target_link_libraries(card-http PRIVATE OpenSSL::Crypto m)
add_executable(card-language language_test.c "${CARD}/card_language.c")
target_include_directories(card-language PRIVATE . "${CARD}")
foreach(target card-core card-ui card-profile card-http card-language)
    target_compile_options(${target} PRIVATE -O1 -g -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
endforeach()
